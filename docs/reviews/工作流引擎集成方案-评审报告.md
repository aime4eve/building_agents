   - 隔离了Camunda内部实现细节
   - 其他服务不直接依赖Camunda API

3. **集成架构符合DDD原则**
   ```
   业务服务 → Feign API → 工作流引擎服务 → Camunda
                 ↑                              ↓
                 └────── 防腐层 ──────────────────┘
   ```

4. **事件驱动架构**
   - ProcessStateChangedEvent
   - SLATimeoutAlertEvent
   - 符合领域事件驱动原则

5. **Phase 1预留接口设计合理**
   - 4个核心接口覆盖基本用例
   - 预留扩展空间

#### ⚠️ 警告项

1. **领域事件命名不符合规范**

   **当前设计：**
   ```java
   public class ProcessStateChangedEvent { }
   public class SLATimeoutAlertEvent { }
   ```

   **DDD规范要求：** `[聚合根名称][过去分词]Event`

   **建议修改为：**
   ```java
   // 流程实例是聚合根
   public class ProcessInstanceStateChangedEvent { }
   public class ProcessInstanceStartedEvent { }
   public class ProcessInstanceCompletedEvent { }

   // SLA监控相关
   public class SLAMonitoringWarningEvent { }
   public class SLAMonitoringBreachedEvent { }
   ```

   ```
   domain层: ProcessInstance（聚合根）

   └── repository: ProcessInstanceRepository

   └── ProcessInstanceMapper（与领域对象转换）
   ```

   **具体调整建议：**
   ```java
   // domain层
   @Value
   class ProcessInstanceId {
       String value;
   }

   enum ProcessInstanceState {
       STARTED, RUNNING, SUSPENDED, COMPLETED, FAILED, CANCELLED
   }

   class ProcessInstance {
       private ProcessInstanceId id;
       private BusinessKey businessKey;
       private ProcessInstanceState state;
       // ...
   }

   // infrastructure层
   @Entity
   @Table(name = "wf_process_instance")
   class ProcessInstancePO {
       @Id
       private String id;
       private String businessKey;
       private String state;
       // ...
   }
   ```

### 聚合根：ProcessInstance

```java
package com.huaquantomt.workflow.domain.model.aggregate;

import com.huaquantomt.workflow.domain.model.domainevent.*;
import com.huaquantomt.workflow.domain.model.valueobject.*;
import lombok.Getter;
import lombok.AccessLevel;

import java.time.LocalDateTime;
import java.util.*;

/**
 * 流程实例聚合根
 */
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class ProcessInstance {

    // 聚合根ID
    private ProcessInstanceId id;

    // 业务键
    private BusinessKey businessKey;

    // 流程定义
    private ProcessDefinitionKey processDefinitionKey;

    // 流程状态
    private ProcessInstanceState state;

    // 租户
    private TenantId tenantId;

    // 启动人
    private UserId startedBy;

    // 当前活动
    private ActivityId currentActivityId;

    // 时间戳
    private LocalDateTime startedAt;
    private LocalDateTime endedAt;
    private LocalDateTime updatedAt;

    // 版本号
    private Long version;

    // 领域事件
    private final List<DomainEvent> domainEvents = new ArrayList<>();

    /**
     * 启动流程实例
     */
    public static ProcessInstance start(
        ProcessDefinitionKey processDefinitionKey,
        BusinessKey businessKey,
        TenantId tenantId,
        UserId startedBy,
        Map<String, Object> variables
    ) {
        ProcessInstance instance = new ProcessInstance();
        instance.id = ProcessInstanceId.generate();
        instance.businessKey = businessKey;
        instance.processDefinitionKey = processDefinitionKey;
        instance.tenantId = tenantId;
        instance.startedBy = startedBy;
        instance.state = ProcessInstanceState.STARTED;
        instance.startedAt = LocalDateTime.now();
        instance.updatedAt = LocalDateTime.now();
        instance.version = 0L;

        // 注册领域事件
        instance.registerDomainEvent(new ProcessInstanceStartedEvent(
            instance.id,
            instance.processDefinitionKey,
            instance.businessKey,
            instance.tenantId,
            instance.startedBy,
            variables,
            instance.startedAt
        ));

        return instance;
    }

    /**
     * 完成任务
     */
    public void completeTask(TaskId taskId, Map<String, Object> variables) {
        TaskId currentTask = getCurrentTask();
        if (!currentTask.equals(taskId)) {
            throw new IllegalStateException("Task is not current activity");
        }

        this.updatedAt = LocalDateTime.now();

        registerDomainEvent(new TaskCompletedEvent(
            this.id,
            taskId,
            this.tenantId,
            variables,
            this.updatedAt
        ));
    }

    /**
     * 完成流程
     */
    public void complete() {
        if (this.state == ProcessInstanceState.COMPLETED) {
            return; // 幂等性
        }

        ProcessInstanceState previousState = this.state;
        this.state = ProcessInstanceState.COMPLETED;
        this.endedAt = LocalDateTime.now();
        this.updatedAt = LocalDateTime.now();

        registerDomainEvent(new ProcessInstanceCompletedEvent(
            this.id,
            this.processDefinitionKey,
            this.businessKey,
            this.tenantId,
            this.startedAt,
            this.endedAt
        ));
    }

    /**
     * 挂起流程
     */
    public void suspend() {
        if (this.state != ProcessInstanceState.RUNNING) {
            throw new IllegalStateException("Only running process can be suspended");
        }

        this.state = ProcessInstanceState.SUSPENDED;
        this.updatedAt = LocalDateTime.now();

        registerDomainEvent(new ProcessInstanceSuspendedEvent(
            this.id,
            this.tenantId,
            this.updatedAt
        ));
    }

    // 领域事件管理
    protected void registerDomainEvent(DomainEvent event) {
        this.domainEvents.add(Objects.requireNonNull(event));
    }

    public List<DomainEvent> getDomainEvents() {
        return Collections.unmodifiableList(domainEvents);
    }

    public void clearDomainEvents() {
        this.domainEvents.clear();
    }
}
```

```java
package com.huaquantomt.workflow.domain.model.valueobject;

import lombok.Value;

@Value
class ProcessInstanceId {
    String value;

    private ProcessInstanceId(String value) {
        if (value == null || value.trim().isEmpty()) {
            throw new IllegalArgumentException("ProcessInstanceId cannot be empty");
        }
        this.value = value;
    }

    public static ProcessInstanceId generate() {
        return new ProcessInstanceId(UUID.randomUUID().toString());
    }

    public static ProcessInstanceId of(String value) {
        return new ProcessInstanceId(value);
    }
}

@Value
class BusinessKey {
    String value;

    private BusinessKey(String value) {
        this.value = Objects.requireNonNull(value, "BusinessKey cannot be null");
    }

    public static BusinessKey of(String value) {
        return new BusinessKey(value);
    }
}

@Value
class ProcessDefinitionKey {
    String value;

    private ProcessDefinitionKey(String value) {
        this.value = Objects.requireNonNull(value, "ProcessDefinitionKey cannot be null");
    }

    public static ProcessDefinitionKey of(String value) {
        return new ProcessDefinitionKey(value);
    }
}

enum ProcessInstanceState {
    STARTED,     // 已启动
    RUNNING,     // 运行中
    SUSPENDED,   // 已挂起
    COMPLETED,   // 已完成
    FAILED,      // 已失败
    CANCELLED    // 已取消
}
```
